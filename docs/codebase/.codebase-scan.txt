# YTDL Flow — Environment Variables
# Copy this file to `.env` and adjust values as needed.
# All variables are optional; defaults are shown in parentheses.

# ── Server ────────────────────────────────────────────────────────────────────
# HTTP server port (1-65535)
# PORT=3000

# HTTP server bind address (default: 127.0.0.1 — loopback only)
# WARNING: Changing this to a LAN IP will cause the Host allowlist to reject
# all requests (DNS rebinding protection). The server is designed for localhost.
# HOST=127.0.0.1

# ── Paths ─────────────────────────────────────────────────────────────────────
# Directory for history.json and other persistent data (default: web/data)
# Override for testing to avoid clobbering real history.
# YTDL_DATA_DIR=

# ── Engine ────────────────────────────────────────────────────────────────────
# Python executable used to spawn the engine (default: python from PATH)
# Point this to a venv interpreter to control dependency resolution.
# PYTHON=python

# Upper bound on queued engine commands when the engine is unavailable (min: 1)
# ENGINE_MAX_PENDING=100

# FFmpeg binary locations — the engine searches these if ffmpeg isn't on PATH.
# FFMPEG_PATH=/path/to/ffmpeg/bin
# FFMPEG_HOME=/path/to/ffmpeg
# DENO_PATH=/path/to/deno

# ── YouTube bot-check bypass ──────────────────────────────────────────────────
# YouTube flags rapid anonymous downloads with "Sign in to confirm you're not
# a bot". Authenticated cookies from a signed-in browser clear the check.
# Use a THROWAWAY YouTube account (automated use risks bans), export cookies
# in Netscape cookies.txt format (browser extension like "Get cookies.txt"),
# then point the engine at it:
# YTDL_COOKIES_FILE=C:\path\to\cookies.txt
# Alternative (reads directly from the local browser profile — no file):
# YTDL_COOKIES_BROWSER=chrome
# Either one is picked up at server start (engine restart required).

# ── Testing / CI ──────────────────────────────────────────────────────────────
# Allow the smoke test to clear history (default: unset — destructive ops skip)
# YTDL_SMOKE_ALLOW_CLEAR=1

# Enable integration tests in the smoke test (default: unset)
# YTDL_INTEGRATION=1
